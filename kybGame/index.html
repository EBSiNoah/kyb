local ReplicatedStorage = game:GetService("ReplicatedStorage") local Players = game:GetService("Players") local RunService = game:GetService("RunService") local Workspace = game:GetService("Workspace") local localPlayer = Players.LocalPlayer local camera = Workspace.CurrentCamera local hitEffectEvent = ReplicatedStorage:WaitForChild("HitEffectEvent") ---------------------------------------------------- -- 1. 화면 흔들림 (Camera Shake) 함수 ---------------------------------------------------- local function shakeCamera(intensity, duration) local startTime = tick() local connection connection = RunService.RenderStepped:Connect(function() local elapsed = tick() - startTime if elapsed >= duration then connection:Disconnect() return end -- 시간이 지남에 따라 감쇠하는 흔들림 효과 local currentIntensity = intensity * (1 - (elapsed / duration)) local offsetX = math.random(-100, 100) / 100 * currentIntensity local offsetY = math.random(-100, 100) / 100 * currentIntensity local offsetZ = math.random(-100, 100) / 100 * currentIntensity camera.CFrame = camera.CFrame * CFrame.Angles( math.rad(offsetX), math.rad(offsetY), math.rad(offsetZ) ) end) end ---------------------------------------------------- -- 2. 히트 파티클 생성 함수 ---------------------------------------------------- local function createHitParticle(position) -- 파티클이 위치할 임시 어태치먼트 생성 local part = Instance.new("Part") part.Size = Vector3.new(0.1, 0.1, 0.1) part.Position = position part.Anchored = true part.CanCollide = false part.Transparency = 1 part.Parent = Workspace local attachment = Instance.new("Attachment") attachment.Parent = part -- ParticleEmitter 설정 local particle = Instance.new("ParticleEmitter") particle.Color = ColorSequence.new(Color3.fromRGB(255, 200, 50)) -- 주황/황금색 particle.Size = NumberSequence.new({ NumberSequenceKeypoint.new(0, 0.5), NumberSequenceKeypoint.new(1, 0) }) particle.Texture = "rbxassetid://243660364" -- 기본 스파크/빛 입자 텍스처 particle.Lifetime = NumberRange.new(0.15, 0.3) particle.Speed = NumberRange.new(15, 30) particle.SpreadAngle = Vector2.new(360, 360) -- 모든 방향으로 발사 particle.LightEmission = 0.8 particle.Parent = attachment -- 한 번에 파티클 방출 particle:Emit(20) -- 연출 후 정리 (메모리 관리) task.delay(0.5, function() part:Destroy() end) end ---------------------------------------------------- -- 3. 서버 이벤트 수신 및 처리 ---------------------------------------------------- hitEffectEvent.OnClientEvent:Connect(function(hitPosition, attacker) -- 모든 플레이어에게 파티클 생성 createHitParticle(hitPosition) -- 내가 직접 때렸을 때만 화면 흔들림 적용 if attacker == localPlayer then -- intensity: 흔들림 강도(도 단위), duration: 지속 시간(초) shakeCamera(2.5, 0.15) end end)
